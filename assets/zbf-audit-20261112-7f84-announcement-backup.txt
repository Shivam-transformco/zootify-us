    <script>
      // Apply saved dismissal before the following header is parsed.
      // This only hides this section; it never edits the page or header offsets.
      (() => {
        const bar = document.getElementById({{ 'ZootifyAnnouncement-' | append: section.id | json }});
        if (!bar) return;
        try {
          const mode = bar.dataset.dismissMode;
          const key = bar.dataset.storageKey;
          let dismissed = mode === 'session' && sessionStorage.getItem(key) === 'dismissed';
          if (mode === 'days') {
            const until = Number(localStorage.getItem(key));
            dismissed = Number.isFinite(until) && until > Date.now();
            if (until && !dismissed) localStorage.removeItem(key);
          }
          if (dismissed) {
            bar.hidden = true;
            const wrapper = bar.closest('.zootify-announcement-section');
            if (wrapper) wrapper.hidden = true;
          }
        } catch (_) { /* Storage can be blocked; the bar still works. */ }
      })();
    </script>
  {% endif %}

            case 'next': this.show(this.index + 1, true); break;

          this.hasInteracted = true;

        });

        this.listen(document, 'zootify:bf-state', () => this.syncCampaign());

          this.stop();
          return;
        }
        this.slides.forEach((slide, i) => {
          const active = i === index;

          slide.setAttribute('aria-hidden', String(!active));
          slide.classList.toggle('is-active', active);
        });
        this.index = index;
        if (manual && this.status && !this.isEditor) {
          const label = (this.dataset.positionLabel || 'Announcement {current} of {total}')

          const countdown = this.slides[index].querySelector('zootify-bf-countdown');
          const text = countdown

          this.status.textContent = label + ': ' + text;
        }
        this.schedule();

        try {
          const key = this.dataset.storageKey;
          if (this.dataset.dismissMode === 'session') return sessionStorage.getItem(key) === 'dismissed';
          if (this.dataset.dismissMode === 'days') {
            const until = Number(localStorage.getItem(key));
            if (Number.isFinite(until) && until > Date.now()) return true;
            if (until) localStorage.removeItem(key);
          }
        } catch (_) {}
        return false;
      }

        try {
          const key = this.dataset.storageKey;
          if (this.dataset.dismissMode === 'session') sessionStorage.setItem(key, 'dismissed');
          if (this.dataset.dismissMode === 'days') {
            const days = Math.max(1, Math.min(30, Number(this.dataset.dismissDays) || 7));
            localStorage.setItem(key, String(Date.now() + days * 86400000));
          }
        } catch (_) {}
        this.stop();
        this.dismissed = true;
        this.hidden = true;
        if (this.wrapper) this.wrapper.hidden = true;
        if (hadFocus && this.wrapper) {
          let sibling = this.wrapper.nextElementSibling;
          while (sibling) {
            const focusable = Array.from(sibling.querySelectorAll('a[href], button:not([disabled]), input:not([disabled]), [tabindex="0"]'))
              .find((node) => node.getClientRects().length && getComputedStyle(node).visibility !== 'hidden');
            if (focusable) { focusable.focus({ preventScroll: true }); break; }
            sibling = sibling.nextElementSibling;
          }
        }
      }

      disconnectedCallback() {
        if (this.startFrame) cancelAnimationFrame(this.startFrame);
        this.stop();
        this.abortController?.abort();
        this.observer?.disconnect();
        this.observer = null;
        this.initialized = false;
      }
    }
    customElements.define(tagName, ZootifyAnnouncement);
  })();
