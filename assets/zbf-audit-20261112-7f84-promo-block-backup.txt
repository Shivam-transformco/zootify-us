{%- doc -%}
  Native collection promotion with its own editable button child.
  @param {string} [display_mode] - Use outside for an editorial collection grid.
{%- enddoc -%}

{%- liquid
  assign promo = block.settings
  assign sale_handle = settings.zbf_sale_handle | default: 'black-friday' | handleize
  assign show_promo = false
  if settings.zbf_enabled and promo.zbf_promo_enabled and collection.handle != sale_handle and template.suffix != 'black-friday'
    assign show_promo = true
    if promo.zbf_promo_scope == 'selected'
      assign included = promo.zbf_promo_collections | map: 'handle'
      unless included contains collection.handle
        assign show_promo = false
      endunless
    endif
    assign excluded = promo.zbf_promo_exclusions | map: 'handle'
    if excluded contains collection.handle
      assign show_promo = false
    endif
  endif
  assign root_tag = 'li'
  if display_mode == 'outside'
    assign root_tag = 'div'
  endif
-%}
{%- capture sale_destination -%}{% render 'zootify-sale-url' %}{%- endcapture -%}
{%- assign destination = promo.zbf_promo_link | default: sale_destination | strip -%}
{%- capture button_markup -%}
  {% content_for 'block', type: '_zootify-campaign-button', id: 'campaign-button', fallback_url: destination, context_name: 'collection-promo', parent_gap_desktop: promo.zbf_promo_gap, parent_gap_mobile: promo.zbf_promo_gap_mobile %}
{%- endcapture -%}
{%- assign button_markup = button_markup | strip -%}
{%- liquid
  if promo.zbf_promo_image == blank and promo.zbf_promo_image_mobile == blank and promo.zbf_promo_heading == blank and promo.zbf_promo_copy == blank and promo.zbf_promo_eyebrow == blank and promo.zbf_promo_accent_text == blank and button_markup == blank
    assign show_promo = false
  endif
-%}
{%- if show_promo -%}
  <{{ root_tag }}
    class="{% if display_mode == 'outside' %}zbf-promo-outside{% else %}zbf-promo-item zbf-promo-item--{{ promo.zbf_promo_layout }} zbf-promo-item--mobile-{{ promo.zbf_promo_mobile_layout | default: 'tile' }}{% endif %}"
    {% unless display_mode == 'outside' %}data-zbf-promo data-page="1"{% endunless %}
    {{ block.shopify_attributes }}
  >
    {% render 'zootify-collection-promo', promo_settings: promo, button_markup: button_markup, display_mode: display_mode, destination: destination %}
  </{{ root_tag }}>
{%- endif -%}
